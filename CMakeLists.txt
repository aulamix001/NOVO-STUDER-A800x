cmake_minimum_required(VERSION 3.22)
project(A800Prototype VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(A800_BUILD_PLUGIN "Build JUCE VST3 technical prototype" ON)
if(BUILD_TESTING)
 add_executable(tape_tests tests/tape_tests.cpp)
 add_test(NAME tape_tests COMMAND tape_tests)
 add_executable(mapping_tests tests/mapping_tests.cpp)
 add_test(NAME mapping_tests COMMAND mapping_tests)
 add_executable(transport_tests tests/transport_tests.cpp)
 add_test(NAME transport_tests COMMAND transport_tests)
endif()
if(NOT A800_BUILD_PLUGIN)
 return()
endif()
set(JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE 7.0.12 checkout")
set(JUCE_HOST_HELPER "" CACHE FILEPATH "Native juceaide executable for cross-compilation")
if(JUCE_HOST_HELPER)
 if(NOT JUCE_SOURCE_DIR)
  message(FATAL_ERROR "JUCE_SOURCE_DIR is required with JUCE_HOST_HELPER")
 endif()
 set(JUCE_MODULES_ONLY ON)
 add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
 include("${JUCE_SOURCE_DIR}/extras/Build/CMake/JUCEUtils.cmake")
 add_executable(juce::juceaide IMPORTED GLOBAL)
 set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${JUCE_HOST_HELPER}")
elseif(JUCE_SOURCE_DIR)
 add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 4f43011b96eb0636104cb3e433894cda98243626)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_binary_data(A800Art SOURCES assets/A800Panel.png)
juce_add_plugin(A800Prototype COMPANY_NAME "Aulamix" BUNDLE_ID "org.aulamix.a800prototype"
 PLUGIN_MANUFACTURER_CODE Aumx PLUGIN_CODE A8p1 FORMATS VST3
 PRODUCT_NAME "A800 Technical Prototype" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE
 NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
 VST3_CATEGORIES Fx VST3_AUTO_MANIFEST FALSE)
target_sources(A800Prototype PRIVATE src/PluginProcessor.cpp src/PluginEditor.cpp src/transport/WindowsTransport.cpp)
target_compile_definitions(A800Prototype PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(A800Prototype PRIVATE juce::juce_audio_utils A800Art PUBLIC juce::juce_recommended_config_flags)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
 juce_add_console_app(processor_tests PRODUCT_NAME "Processor tests")
 target_sources(processor_tests PRIVATE tests/processor_tests.cpp src/PluginProcessor.cpp src/PluginEditor.cpp src/transport/WindowsTransport.cpp)
 target_compile_definitions(processor_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(processor_tests PRIVATE juce::juce_audio_utils A800Art)
 add_test(NAME processor_tests COMMAND processor_tests)
endif()

option(A800_BUILD_TRANSPORT_TESTHOST "Build isolated Win32 transport integration test" OFF)
if(A800_BUILD_TRANSPORT_TESTHOST AND WIN32)
 add_executable(transport_testhost tests/windows_transport_testhost.cpp src/transport/WindowsTransport.cpp)
 set_target_properties(transport_testhost PROPERTIES OUTPUT_NAME "Studio One" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/transport-testhost")
 target_link_libraries(transport_testhost PRIVATE user32)
endif()
